cmake_minimum_required(VERSION 3.20)
project(termous_video_sdk LANGUAGES C CXX)

if(MSVC OR CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Use a native GCC/Apple Clang toolchain; MSVC is not supported")
endif()
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "SDK builds require Release mode")
endif()
if(NOT TERMOUS_TARGET MATCHES "^(windows/amd64|linux/amd64|darwin/arm64)$")
  message(FATAL_ERROR "Unsupported SDK target")
endif()
string(TOLOWER "${CMAKE_SYSTEM_NAME}" TERMOUS_HOST_SYSTEM)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" TERMOUS_HOST_ARCH)
if(TERMOUS_HOST_ARCH MATCHES "^(x86_64|amd64)$")
  set(TERMOUS_HOST_ARCH amd64)
elseif(TERMOUS_HOST_ARCH MATCHES "^(arm64|aarch64)$")
  set(TERMOUS_HOST_ARCH arm64)
endif()
if(NOT TERMOUS_TARGET STREQUAL "${TERMOUS_HOST_SYSTEM}/${TERMOUS_HOST_ARCH}" OR
   NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "SDK target must match the native 64-bit toolchain")
endif()
foreach(language C CXX)
  if((WIN32 AND NOT CMAKE_${language}_COMPILER_ID STREQUAL "GNU") OR
     (APPLE AND NOT CMAKE_${language}_COMPILER_ID STREQUAL "AppleClang") OR
     (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT CMAKE_${language}_COMPILER_ID STREQUAL "GNU"))
    message(FATAL_ERROR "Both SDK compilers must be MinGW/GCC or Apple Clang")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(H264_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../build/sources/openh264")
set(YUV_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../build/sources/libyuv")
include(cmake/openh264.cmake)
include(cmake/libyuv.cmake)
install(TARGETS openh264 yuv ARCHIVE DESTINATION lib)
install(DIRECTORY "${H264_ROOT}/codec/api/wels" DESTINATION include FILES_MATCHING PATTERN "*.h")
install(DIRECTORY "${YUV_ROOT}/include/" DESTINATION include FILES_MATCHING PATTERN "*.h")
